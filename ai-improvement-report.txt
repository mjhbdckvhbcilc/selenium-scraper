def scrape_website(url: str) -> List[str]:
    """
    Scrape a website for headlines.

    Args:
        url: The URL of the website to scrape.

    Returns:
        A list of headlines found on the website.
    """
    driver = webdriver.Chrome()  # Ensure you are using a trusted version of chromedriver
    driver.get(url)
    headlines = driver.find_elements(By.TAG_NAME, 'h1')  # Example usage
    driver.quit()

    return [headline.text for headline in headlines]
