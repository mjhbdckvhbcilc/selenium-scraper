def scrape_data(url: str):
    """Scrape data from a given URL using Selenium WebDriver."""
    # Improved variable naming
    driver = webdriver.Chrome()  # Ensure Chrome is installed and accessible
    driver.get(url)

    # Extract and return data
    title = driver.find_element_by_tag_name('h1').text
    description = driver.find_element_by_class_name('description').text
    data = {
        'title': title,
        'description': description,
    }
    driver.quit()  # Ensure to close the browser

    # Return the scraped data
    return data

# Example usage
if __name__ == '__main__':
    url = 'https://example.com'
    scraped_data = scrape_data(url)
    print(f'Scraped data: {scraped_data}')
