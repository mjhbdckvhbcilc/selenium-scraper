def fetch_data(url):
    """
    Fetches HTML content from a given URL.
    """
    driver = initialize_driver()  # Ensure driver is properly initialized with input validation.
    driver.get(url)
    content = driver.page_source
    return content

def initialize_driver():
    try:
        driver = webdriver.Chrome()  # Ensure appropriate driver is loaded and configured
        driver.set_page_load_timeout(10)  # Add a timeout to prevent infinite waits
        return driver
    except Exception as e:
        raise Exception("Failed to initialize the WebDriver: ", str(e))
