```python
def scrape_data() -> str:
    """Scrape the data from a specified URL using Selenium.
    """
    url = 'https://example.com'
    browser = webdriver.Chrome()  # Ensure the correct WebDriver is installed
    browser.get(url)
    data = browser.find_element(By.ID, 'data').text
    browser.quit()  # Ensure browser is closed after scraping
    return data
```
