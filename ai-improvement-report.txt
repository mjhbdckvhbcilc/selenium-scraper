```python
# selenium_scraper.py
import time
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys

def scrape_website(url):
    """
    Scrape a website using Selenium and return the content.
    """
    # Initialize the webdriver. Here we use Chrome, but you can choose any browser.
    driver = webdriver.Chrome()

    try:
        # Navigate to the URL
        driver.get(url)

        # Wait for the content to load (can be replaced with explicit waits for more complex scenarios)
        time.sleep(2)

        # Use XPath to find elements
        element = driver.find_element(By.XPATH, "//div[@class='content']")
        
        # Extract the content
        content = element.text

        # Return the extracted content
        return content

    finally:
        # Close the browser
        driver.quit()

if __name__ == "__main__":
    url = "http://example.com"  # Replace with the URL you want to scrape
    result = scrape_website(url)
    print(result)
```
