```javascript
// scraping.js
function scrapeWebsite(url) {
    const browser = new webdriver.Builder()
        .forBrowser('chrome') // or specify another browser like 'firefox', etc.
        .build();
    
    browser.get(url)
      .then(() => {
          console.log("Navigated to: " + url);
          
          let pageSource = browser.getPageSource();
        
          if (pageSource) {
              const elements = document.querySelectorAll('div.content');
              
              Array.from(elements).forEach(element => {
                  const title = element.querySelector('h1').innerText;
                  const content = element.querySelector('p').innerText;

                  console.log(`Title: ${title}, Content: ${content}`);
              });
          }
      })
      .finally(() => browser.quit());
}

// Example usage:
scrapeWebsite('http://example.com');
```
